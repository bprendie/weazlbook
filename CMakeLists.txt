cmake_minimum_required(VERSION 3.20)
project(weazlbook_native C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
find_package(ZLIB REQUIRED)
file(GLOB PDFIO_SOURCES CONFIGURE_DEPENDS vendor/pdfio/pdfio-*.c)
add_library(pdfio STATIC ${PDFIO_SOURCES} vendor/pdfio/ttf.c)
target_include_directories(pdfio PUBLIC vendor/pdfio native)
target_link_libraries(pdfio PUBLIC ZLIB::ZLIB m)
add_library(wbcore native/pdf.c native/budget.c native/core.cpp native/markdown.cpp native/epub.cpp vendor/pugixml/pugixml.cpp)
target_include_directories(wbcore PUBLIC native vendor/pdfio vendor/pugixml)
target_link_libraries(wbcore PUBLIC pdfio)
add_executable(wb-native native/cli.cpp)
target_link_libraries(wb-native PRIVATE wbcore)
enable_testing()
add_executable(core-test native/test_core.cpp)
target_link_libraries(core-test PRIVATE wbcore)
add_test(NAME core COMMAND core-test)
